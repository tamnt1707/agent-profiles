#!/bin/zsh
# Managed by Claude Profiles. Rewritten on every save.
# Tạo bởi Claude Profiles cho profile "Test". File này được ghi đè mỗi lần lưu.

# Bỏ cấu hình nhà cung cấp còn sót trong shell để profile không bị lẫn
unset ANTHROPIC_API_KEY ANTHROPIC_AUTH_TOKEN ANTHROPIC_BASE_URL ANTHROPIC_MODEL \
  ANTHROPIC_DEFAULT_OPUS_MODEL ANTHROPIC_DEFAULT_SONNET_MODEL ANTHROPIC_DEFAULT_HAIKU_MODEL \
  ANTHROPIC_SMALL_FAST_MODEL

__k="$(/usr/bin/security find-generic-password -s 'claude-profiles' -a 'test' -w 2>/dev/null)"
if [[ -z "$__k" ]]; then
  echo "Chưa có API key cho profile 'Test' trong Keychain. Mở app Claude Profiles để nhập." >&2
  exit 1
fi
export ANTHROPIC_API_KEY="$__k"
unset __k

export ANTHROPIC_BASE_URL='https://example.com/anthropic'
export ANTHROPIC_MODEL='m-1'
export ANTHROPIC_DEFAULT_OPUS_MODEL='o-1'
export ANTHROPIC_DEFAULT_SONNET_MODEL='s-1'
export ANTHROPIC_DEFAULT_HAIKU_MODEL='h-1'
export API_TIMEOUT_MS='600000'

exec claude "$@"
